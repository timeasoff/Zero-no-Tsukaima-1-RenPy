/**
 * ツンデレメータ
 */
class Tundele {

    before = null;
    after = null;
    state = null;
    now   = null;
    time = null;

    // ベースレイヤ
    baseLayer = null;
    // 針のレイヤ
    arrowLayer = null;

    // アップダウン表示用
    paraUpDownLayer = null;
    // ツンデレ表示用
    paraTunDereLayer = null;
    
    // 設定
    dispTime  = 500.0; // 表示にかかる時間
    waitTime1 = 500.0; // 移動前の待ち時間
    moveTime  = 800.0; // 針の移動時間
    waitTime2 = 1500.0; // 移動後の待ち時間

    // ツンデレメータ表示用
    xpos = 400;
    ypos = 20;
    arrowydiff = 16;
    arrowxdiff = 40-6;
    arrowwidth = 156;

    // アップ・ダウン表示用
    paraXpos = 320;
    paraYpos = 100;
    paraXpos2 = 40;
    paraYpos2 = 90;
    paraYdiff = 20;

    seId = null;
    skipFlag = false;
    
    constructor(before, after) {
        this.before = before;
        this.after  = after;

        baseLayer = Layer(true);
        baseLayer.loadImage(tndlBaseImageId);
        baseLayer.setLevel(3);
        baseLayer.setPos(xpos, ypos);

        arrowLayer = Layer(true);
        arrowLayer.loadImage(tndlArrowImageId);
        arrowLayer.setLevel(2);

        paraTunDereLayer = Layer(true);
        paraTunDereLayer.loadImage(before < after ? paraDereImageId : paraTunImageId);
        paraTunDereLayer.setLevel(2);
        paraTunDereLayer.setPos(paraXpos, paraYpos);
        paraTunDereLayer.show(false);

        paraUpDownLayer = Layer(true);
        paraUpDownLayer.loadImage(paraUpImageId);
        paraUpDownLayer.setLevel(2);
        paraUpDownLayer.setPos(paraXpos + paraXpos2, paraYpos + paraYpos2);
        paraUpDownLayer.show(false);

        // メータ制御
        setPos(before);
        setOpacity(0);
        setShow(true);

        state = 0;
        now = 0;
        time = dispTime;

        seId = before < after ? paraUpSEId : paraDownSEId;
    }

    /**
     * ツンデレメータ透明度
     * @param opacity 透明度
     */
    function setOpacity(opacity) {
        local op = (128 * opacity / 100).tointeger();
        baseLayer.setColor(100,100,100,op);
        arrowLayer.setColor(100,100,100,op);
    }


    /**
     * ツンデレメータ透明度
     * @param opacity 透明度
     */
    function setUpOpacity(opacity) {
        local op = (128 * opacity / 100).tointeger();
        paraUpDownLayer.setColor(100,100,100, op);
        paraUpDownLayer.setPos(paraXpos + paraXpos2, (paraYpos + paraYpos2 - paraYdiff * opacity / 100).tointeger());
    }

    /**
     * ツンデレメータ矢印位置
     * @param pos 位置指定
     */
    function setPos(pos) {
        local x = (xpos + pos * arrowwidth / 256 + arrowxdiff).tointeger();
        arrowLayer.setPos(x, ypos + arrowydiff);
    }

    /**
     * ツンデレメータ表示状態
     * @param show 表示状態
     */
    function setShow(show) {
        baseLayer.show(show);
        arrowLayer.show(show);
    }

    /**
     * 状態更新
     * @param diff 経過時間
     */
    function update(diff) {
        if (skipFlag || isSkip()) {
            now = time;
        } else {
            now += diff;
        }
        switch (state) {
        case 0:
            // 表示処理
            if (now < time) {
                setOpacity(100 * (1.0 -  (time - now) / dispTime));
            } else {
                setOpacity(100);
                time += waitTime1;
                state++;
                // 効果音再生
                if (!(skipFlag ||isSkip())) {
                    if (seId != null) {
                        MusicStop(1, 0);
                        MusicPlay(1, seId, volumeCalc(100), timeCalc(0));
                    }
                }
            }
            break;
        case 1:
            // 待ち
            if (now < time) {
            } else {
                time += moveTime;
                state++;
                // 表示
                setUpOpacity(0);
                paraUpDownLayer.show(true);
                paraTunDereLayer.show(true);
            }
            break;
        case 2:
            // 針の移動
            if (now < time) {
                setPos(before + (after - before) * (1.0 -  (time - now) / moveTime));
                setUpOpacity(20 + 80 * (1.0 -  (time - now) / moveTime));
            } else {
                setPos(after);
                setUpOpacity(100);
                time += waitTime2;
                state++;
            }
            break;
        case 3:
            // 待ち
            if (now < time) {
            } else {
                time += dispTime;
                state++;
                // 消去
                paraUpDownLayer.show(false);
                paraTunDereLayer.show(false);
            }
            break;
        case 4:
            // 消去処理
            // 表示処理
            if (now < time) {
                setOpacity(100 * (time - now) / dispTime);
            } else {
                setOpacity(0);
                state++;
            }
            break;
        case 5:
            // 終了
            paraUpDownLayer.show(false);
            paraTunDereLayer.show(false);
            setShow(false);
            return false;
            break;
        }
        return true;
    }

    /**
     * 強制終了
     */
    function done() {
        skipFlag = true;
    }
};

/**

function msgshow()
{
    if (msg == null) {
        msg = MessageWindow(48, 336, 1);
    }
    if (msgWinState) {
        if (msgText != null) {
            msg.write(msgText);
        } else {
            msg.clear();
        }
        if (msgName != null) {
            msg.ShowNamePlate(msgName);
        } else {
            msg.ShowNamePlate(0);
        }
        msg.showCursor(msgCursor);
        msg.show(true);
    } else {
        msg.show(false);
    }
}

/**
 * メッセージ窓消滅
 */
function msghide()
{
    msg = null;
}

/**
 * メッセージ窓表示
 */
function msgon()
{
    msgWinState = true;
    msgshow();
}

/**
 * メッセージ窓消去
 */
function msgoff()
{
    msgWinState = false;
    msgText = null;
    msgName = null;
    msgCursor = false;
    if (msg != null) {
        msg.clear();
        msg.showCursor(false);
        msg.ShowNamePlate(0);
        msg.show(msgWinState);
    }
}

/**
 * 画面フェード処理用のレイヤ
 */
class ScreenLayer extends BasicLayer
{
    // 選択したモード
    mode = 0;
    // 時間指定
    time = 0;
    // swap フラグ
    swap = false;

    // 実行時間
    actionTime = null;
    
    constructor() {
        ::BasicLayer.constructor(true);
    }

    /**
     * レイヤの差し替え処理
     */
    function swapLayers() {
        if (swap) {
            foreach (idx,val in objects) {
                val.backDone();
            }
            swap = false;
            syncFore = true;
        }
    }
    
    /**
     * フェード処理を開始する
     */
    function start(mode, time, swap) {
        this.mode = mode;
        this.time = time;
        this.swap = swap;
        actionTime = 0;
        if (mode == 2) {
            // フェードインの場合はまずスワップ
            swapLayers();
        }
    }

    /**
     * 処理ハンドラ
     */
    function handler(now, stopFlag) {
        //print("screen:" + now + "\n");
        local done = false;
        if (isSkip() || stopFlag) {
            now = time;
        }
        if (now >= time) {
            done = true;
        }
        switch (mode) {
        case 1:
            // fadeout
            setOpacity(100 * now / time);
            break;
        case 2:
            // fadein
            setOpacity(100 - 100 * now / time);
            break;
        case 3:
            // fadeout/in
            {
                local time2 = time / 2;
                if (now < time2) {
                    // fadeout
                    setOpacity(100 * now / time2);
                } else {
                    // fadein
                    swapLayers();
                    local n = now - time2;
                    setOpacity(100 - 100 * n / time2);
                }
            }
            break;
        }
        if (done) {
            swapLayers();
            actionTime = null;
            if (mode ==2 || mode == 3) {
                screenLayer.setShow(false);
            }
            trigger("screen");
        }
    }

    function update(diff) {
        if (actionTime != null) {
            actionTime += diff;
            handler(actionTime, false);
        }
    }

    function stop() {
        handler(actionTime, true);
        actionTime = null;
    }
}

// 黒画像と白画像
screenLayer <- null;
blackImageId <- 0;
whiteImageId <- 0;
lastImageId <- null;

/**
 * スクリーン画像を表示
 */
function screenShow(image, opacity)
{
    if (screenLayer == null) {
        screenLayer = ScreenLayer();
        screenLayer.setPos(0,0);
        screenLayer.setLevel(1);
        if (image != null) {
            lastImageId = image;
        }
        if (lastImageId == null) {
            lastImageId == blackImageId;
        }
        screenLayer.loadImage(lastImageId);
    } else {
        if (image != null && lastImageId != image) {
            lastImageId = image;
            screenLayer.loadImage(lastImageId);
        }
    }
    screenLayer.fore();
    screenLayer.setOpacity(opacity);
    screenLayer.setShow(true);
}

/**
 * スクリーン処理の停止
 */
function screenStop()
{
    if (screenLayer != null) {
        screenLayer.stop();
    }
}

/**
 * スクリーン処理終了待ち
 */
function waitScreen(canskip)
{
    if (screenLayer == null || screenLayer.actionTime == null) {
        return;
    }
    if (canskip) {
        if (isSkip()) {
            screenStop();
            return;
        } 
        addWait({
            click = function() { screenStop(); },
            screen = function() {}
        }, null);
    } else {
        addWait({
            screen = function() {}
        }, null);
    }
}
    
/**
 * スクリーン画像フェードアウト
 */
function fadeOut(image, time)
{
    screenShow(image, 0);
    screenLayer.start(1, time, false);
    waitScreen(true);
    suspend(WAIT);
}

}

/**
 * 再生開始。指定されたシーンから再生開始します。
 * @param scene 再生するシーン
 * @param lineNo 指定した行から再生開始
 */
function start(name, lineNo)
{
	// サウンド系を停止
	MusicStop(0, 0);
	MusicStop(1, 0);

    // ファンクションの強制破棄
    killFunc();

    //print("scene start:" + name + " : " + sceneThread  + "\n");
    _reset();
    autoMode = false;
    skipMode = 0;
    prevSkipMode = null;
    paused = false;
    
    scene       = name;
    targetLine  = lineNo;
    currentLine = 0;
    nextScene   = null;
    sceneThread = getThread(scene);
    printInfo();
}

/**
 * ジャンプ実行中かどうか
 */
function isJump()
{
    return targetLine != null && currentLine < targetLine;
}

/**
 * ジャンプ実行中かどうか
 */
function isJump2()
{
    return targetLine != null && currentLine <= targetLine;
}

/**
 * スキップ中かどうか
 */
function isSkip()
{
    return skipMode != 0;
}

/**
 * 再生停止
 */
function stop() {
    scene = null;
    targetLine = null;
    currentLine = 0;
    nextScene = null;
    sceneThread = null;
    _reset();
    msghide();
}

{

    }

/**
 * 一時停止。一時的にシーンプレイヤーの動作を停止状態にします
 */
function pause()
{
    if (!paused) {
        _vibrate(0);
        stopVoice();
        msghide();
        {
            // 黒レイヤ生成
            local black = Layer(true);
            black.loadImage(blackImageId);
            black.setPos(0,0);
            black.setLevel(-1);
            black.setColor(100,100,100,0); // 透明
            black.show(true);

            // 全オブジェクトを停止
            foreach (idx,val in objects) {
                val.pause();
            }

            // ファンクションの強制破棄
            killFunc();
        }
        paused = true;
    }
}

/**
 * 待ち情報の追加
 * @param until 待ち情報テーブル (待ち解除時関数およびパラメータ)
 * @param timeout タイムアウト指定
 */
function addWait(until, timeout)
{
    // 待ち情報のコピー
    foreach (idx,val in until) {
        //print("addWait " + idx + ":" + val + "\n");
        waitUntil[idx] <- val;
    }
    // タイムアウト登録
    if (timeout != null) {
        if (timeout <= 0) {
            timeout = 1;
        }
        ::timeout = timeout;
    }
}

function clearWait()
{
    // 待ち情報の廃棄
    foreach (idx,val in waitUntil) {
        delete waitUntil[idx];
    }
}

/**
 * 外部トリガ通知
 * 外部からシーンプレイヤーに対して状態の変更を通知します。
 * 例：クリックされた、アニメーションが終了したなど
 * 「処理待ち」の状態の解除のための機構です
 * @param trigger トリガの名前
 */
function trigger(trigger)
{
    //print("trigger:" + trigger + "\n");
    if (trigger in waitUntil) {
        local func = waitUntil[trigger];
        //print("trigger wait clear:" + (typeof func) + "\n");
        if (func != null && typeof func == "function") {
            local argname = trigger + "_arg";
            local arg = null;
            if (argname in waitUntil) {
                arg = waitUntil[argname];
            }
            if (arg != null) {
                func(arg);
            } else {
                func();
            }
        }
        clearWait();
    }
}

/**
 * 各種時間待ち更新
 */
function updateTime(diff)
{
    // タイムアウト処理
    if (timeout != null && timeout > 0) {
        timeout -= diff;
        if (timeout <= 0) {
            timeout = null;
            trigger("timeout");
        }
    }
    
    // 会話用更新
    if (talkNextTime > 0) {
        talkNextTime -= diff;
	}

    // ボイス用更新
    if (talkVoiceLength > 0) {
        talkVoiceLength -= diff;
    }
}

// 今アクションが実行中かどうか
currentAction <- false;

/**
 * 各種時間待ち更新
 */
function updateObject(diff)
{
    // オブジェクトの状態更新
    local prevCurrentAction = currentAction;
    currentAction = false;
    foreach (idx,val in objects) {
        val.update(diff);
        if (val.isAction()) {
            currentAction = true;
        }
    }
    // 全アクション終了
    if (prevCurrentAction == true && currentAction == false) {
        trigger("action");
    }
    
    // 特殊ファンクション実行
    if (funcs.len() > 0) {
        for (local i=funcs.len() - 1; i>=0; i--) {
            local func = funcs[i];
            if (!func.update(diff)) {
                funcs.remove(i);
            }
        }
        if (funcs.len() == 0) {
            trigger("func");
        }
	}

    // 画面合成用レイヤ
    if (screenLayer != null) {
        screenLayer.update(diff);
    }
}

// 定数定義
WAIT   <- 1;  // プレイヤーがなんらかの待ち状態になっていることの通知。
CHANGE <- 2;  // 状態変更。シーンが切り替わった
DONE   <- 3;  // 完了 シーンの末尾に到達して実行終了したことを通知します。
SELECT <- 4;  // 選択肢処理実行
MOVE   <- 5;
DATE   <- 6;
COFFEE <- 7;
BATTLE <- 8;
SYSSAVE <- 9;
MOVIE <- 10;

function isFunc()
{
    return funcs.len() > 0;
}

// ファンクション系の強制終了
function stopFunc()
{
    for (local i=funcs.len() - 1; i>=0; i--) {
        funcs[i].done();
    }
}

// ファンクション系の強制破棄
function killFunc()
{
    if (funcs.len() > 0) {
        for (local i=funcs.len() - 1; i>=0; i--) {
            local func = funcs[i];
            funcs.remove(i);
        }
        trigger("func");
    }
}

 * 機能終了待ち

        } else {
            addWait({
                func = function() {},
                timeout = function() { stopFunc();},
            }, waittime);
        }
        suspend(WAIT);
    }
}

/**
 * メイン呼び出し
 * @param diff 前回呼び出しからの経過時間 (ms)
 * @return 呼び出し完了後のシーンプレイヤーの状態
 */
function sceneMain(diff)
{
    // 時間更新処理
    updateTime(diff);

    // ポーズ中
    if (paused) {
        return WAIT;
	}

    // 保有しているオブジェクトがスレッドでない場合は終了
    if (sceneThread == null || typeof sceneThread != "thread") {
        return DONE;
    }

    // オブジェクト状態更新
    updateObject(diff);
    
    // 処理待ちがある場合
    if (!msgWinShow ||
        waitUntil.len() > 0 ||
        talkNextTime > 0) {
        return WAIT;
    }

    // 安定状態から抜ける
    stable = false;
    
    // 会話処理中
    if (talkText != null) {
        msgshow();
        if (talkName != null) {
            msgname(talkName);
            talkName = null;
        }
        if (talkVoice != null) {
            if (!isSkip() && talkVoice >= 0) {
                startVoice(talkVoice);
            } else {
                stopVoice();
            }
            talkVoice = null;
        }
        local speed = getCharSpeed();
        if (!isSkip() && speed > 0 && talkCount < talkText.len()) {
            talkCount += 2;
            if (talkCount < talkText.len()) {
                local text = talkText.slice(0,talkCount);
                msgwrite(text);
            } else {
                msgwrite(talkText);
            }
            talkNextTime = speed;
        } else {
            // メッセージ全表示完了
            msgwrite(talkText);
            if (!isSkip()) {
                if (autoMode) {
                    startAuto();
                } else {
                    addWait({
                        click = function() { stopVoice(); talkWait= false;}
                    }, null);
                }
                msgwait();
                stable = true;
            } else {
                stopVoice();
            }
            talkText = null;
            talkNextTime = 0;
		}
        // 会話表示処理中
        return WAIT;
	}

    //print("thread status:" + sceneThread.getstatus() + "\n");

    local ret;
    try {
        if (sceneThread.getstatus() == "suspended") {
            ret = sceneThread.wakeup();
        } else {
            ret = sceneThread.call();
        }
    } catch (e) {
        print("error " + scene + ":" + currentLine + ":" + e + "\n");
    }

    if (sceneThread.getstatus() == "idle") {
        // 終了した場合
        //print("done sceneThread next:" + nextScene + "\n");
        if (nextScene != null) {
            // 新たなシーン
            scene = nextScene;
            targetLine  = null;
            currentLine = 0;
            sceneThread = getThread(nextScene);
            printInfo();
            nextScene = null;
            _vibrate(0);
            prevSkipMode = null;
            return CHANGE;
        }
    } else if (sceneThread.getstatus() == "suspended") {
        // 中断した場合
        //print("sceneThread suspended\n");
        if (ret != null) {
            return ret;
        }
    }

    // 全終了
    sceneThread = null;
    _vibrate(0);
    return DONE;
}

    if (skipMode == 1 || skipMode == 2) {

function onSkip(param)
{
    if (!paused && msgWinShow) {
        if (skipMode == 1) {
            //print("スキップ解除\n");
            skipMode = 0;
        } else if (skipMode == 0) {
            if (prevSkipMode != null) {
                if (prevSkipMode == 0) {
                    if (isAllSkip() || getReaded(scene) >= currentLine) {
                        prevSkipMode = 1;
                        stopVoice();
                    }
                }
                skipMode = 3;
            } else {
                onClick();
                if (isAllSkip() || getReaded(scene) >= currentLine) {
                    //print("スキップ開始\n");
                    skipMode = 1;
                    stopVoice();
                } else {
                    //print("未読でスキップ開始できず\n");
                }
            }
        }
    }
}

{
    if (!paused && msgWinShow) {
        if (skipMode < 2) {
            print("強制スキップON\n");
            cancelAuto();
            onClick();
            skipMode = 2;
            stopVoice();
        }
    }
}

{
    if (skipMode == 2) {
        print("強制スキップOFF\n");
        skipMode = 0;
    }
}

function onAuto(param)
{
    if (!paused && msgWinShow) {
        if (autoMode) {
            cancelAuto();
        } else {
            autoMode = true;
            // もし会話待ちならそのまま自動待ちに変更
            if (talkWait) {
                startAuto();
            }
        }
    }
}

// 自動待ち処理
function startAuto()
{
    local lastLength;
    if (Talkingnow(0) && talkVoiceLength > 0) {
        lastLength  = talkVoiceLength;
    } else {
        lastLength  = 0;
    }
    lastLength += getAutoWait();
    addWait({
        click = function() { stopVoice(); talkWait = false; },
        timeout = function() { stopVoice(); talkWait = false; },
    }, lastLength);
}

{
    if (!paused) {
        cancelAuto();
        cancelSkip();
        // 表示してる場合のみ有効
        if (msgWinState) {
            if (msgWinShow) {
                // 窓を消去して ON 指令がくるまで停止する
                msghide();
                msgWinShow = false;
            } else {
                // 窓を表示して待ちの解除
                msgshow();
                msgWinShow = true;
            }
        }
    }
}

/**

function back()
{

        foreach (idx,val in objects) {

{
    foreach (idx,val in objects) {
        val.backDone();
    }
    syncFore = true;
}

/**
 * 画面切り替え処理
 * @param name トランジション種類
 * @param time 時間
 */
function trans(name, time)
{
    if (!isJump()) {
        if (name == null) {
            name = "";
        }
        if (time == null) {
            time = 0;
        }
        if (name == "whiteout") {
            screenShow(whiteImageId, 0);
            screenLayer.start(1, time, true);
            waitScreen(true);
        } else if (name == "blackout") {
            screenShow(blackImageId, 0);
            screenLayer.start(1, time, true);
            waitScreen(true);
        } else if (name == "fadein") {
            screenShow(null, 100);
            screenLayer.start(2, time, true);
            waitScreen(true);
        } else if (name == "whitein") {
            screenShow(whiteImageId, 100);
            screenLayer.start(2, time, true);
            waitScreen(true);
        } else if (name == "blackin") {
            screenShow(blackImageId, 100);
            screenLayer.start(2, time, true);
            waitScreen(true);
        } else if (name == "whiteoutin") {
            screenShow(whiteImageId, 0);
            screenLayer.start(3, time, true);
            waitScreen(true);
        } else if (name == "blackoutin") {
            screenShow(blackImageId, 0);
            screenLayer.start(3, time, true);
            waitScreen(true);
        } else {
            swapStart(name, time.tointeger());
            waitTrans(true);
        }
        suspend(WAIT);
    }
}

/**
 * 画像のフェードロードの終了待ち
 */
function waitLoad(canskip)
{
    if (!isJump()) {
        // トランジション待ち
        if (!isDriveImage()) {
            return;
        }
        if (canskip) {
        if (isSkip()) {
            loadStop();
            return;
        } 
            addWait({
                click = function() { loadStop(); },
                trans = function() {}
            }, null);
        } else {
            addWait({
                trans = function() {}
            }, null);
        }
        suspend(WAIT);
    }
}

// 配列から適当に選ぶ
function choice(array) {
    return array[random(array.len())];
}

// シーン選択
function getTunDeleScene(tun, event, restrict)
{
    // レベル2 を探す
    local level1 = [];
    local level2 = [];
    for (local i=0;i<6;i++) {
        // 制限処理
        if (restrict) {
            if (tun) {
                if (i == 2 || i == 3) {
                    continue;
                }
            } else {
                if (i == 0 || i == 3) {
                    continue;
                }
            }
        }
        switch (event[i]) {
        case 0:
            level1.append(i);
            break;
        case 1:
            // 確率４倍
            level1.append(i);
            level1.append(i);
            level1.append(i);
            level1.append(i);
            break;
        case 2:
            if (!(restrict && tun && (i==1 || i==4))) {
                level2.append(i);
            }
            break;
        }
    }

    // レベル2 があれば優先再生
    if (level2.len() > 0 || level1.len() > 0) {
        zero_tndl_no = choice(level2.len() > 0 ? level2 : level1);
        return (tun ? tnEventId: dlEventId) + 2 + (zero_tndl_no * 3) + event[zero_tndl_no];
    } else {
        return 0;
    }
}

/**
 * @return ツンイベントシーンID
 */
function getTunScene(restrict)
{
    return getTunDeleScene(true, zero_tun_event, restrict);
}

/**
 * @return デレイベントシーンID
 */
function getDeleScene(restrict)
{
    return getTunDeleScene(false, zero_dele_event, restrict);
}

/**
 * 移動処理実行
 * @param 
 */
function date(done, restrict, bgm)
{
    //print("デート実行\n");
    zero_tndl_flag = 0;
    if (zero_tundele < 40) {
        // ツンイベント発生
        zero_tndl_scene = getTunScene(restrict);
        if (zero_tndl_scene != 0) {
            zero_tndl_flag = 1;
            zero_tndl_done = done;
            //resetSelect(tnEventId, tnEventId + 1);
            resetSelect2(zero_tndl_scene);
        }
    } else if (zero_tundele >= 211) {
        // デレイベント発生
        zero_tndl_scene = getDeleScene(restrict);
        if (zero_tndl_scene != 0) {
            zero_tndl_flag = 2;
            zero_tndl_done = done;
            //resetSelect(dlEventId, dlEventId + 1);
            resetSelect2(zero_tndl_scene);
        }
    }
    // 選択肢登録
    ::DateClear(zero_tndl_flag != 0);
    for (local i=0; i<selectInfos.len(); i++) {
        local info = selectInfos[i];
        ::DateAdd(info.chr);
    }
    stable = true;
    if (bgm >= 0) {
        MusicPlay(0, bgm, volumeCalc(100), timeCalc(500));
    }
    suspend(DATE);
}

function dateDone(param)
{
    print("dateDone:" + param + "\n");
    // BGM 停止
    if (zero_tndl_flag != 0) {
        if (zero_tndl_flag == 1) {
            zero_tun_event[zero_tndl_no]++;
        } else {
            zero_dele_event[zero_tndl_no]++;
        }
        zero_tundele_count++;
        zero_tndl_flag = 0;
    }
    MusicPlay(0, 0, 0, timeCalc(500));
    if (selectInfos.len() > 0) {
        if (param < selectInfos.len()) {
            local info = selectInfos[param];
            if ("eval" in info && info.eval != null) {
                eval(info.eval, null);
            }
            if ("next" in info && info.next != null) {
                next(info.next, null);
            }
        }
        selectInfos = [];
    }
}

/**

 */
function isMax(n)
{
    local v = zero_likes[n];
    for (local i=0;i<6;i++) {
        if (i != n && zero_likes[i] > v) {
            return false;
        }
    }
    return true;
}

 * ルート決定
 */

{
    addLine();
    if (!isJump2()) {
        if (zero_tundele_count >= 4) {
            zero_route = 1;
        } else {
            if (zero_tundele_count >= 4) {
                zero_route = 1; // ルイズ
            } else if (isMax(5)) { // ハルナ
                zero_route = 6;
            } else if (isMax(3)) { // シエスタ
                zero_route = 4; 
            } else if (isMax(4)) { // アンリエッタ
                zero_route = 5;
            } else if (isMax(2)) { // タバサ
                zero_route = 3;
            } else if (isMax(1)) { // キュルケ
                zero_route = 2;
            } else {
                zero_route = 1; // ルイズ
            }
        }
    }
}
